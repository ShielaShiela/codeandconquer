C++ Arrays vs Python Lists

C++ 배열과 Python 리스트 비교

Introduction to Arrays and Lists

배열과 리스트 소개

🔵 C++ Arrays / C++ 배열

  • ✓ Fixed size / 고정 크기
  • ✓ Contiguous memory / 연속된 메모리
  • ✓ Same data type / 동일한 데이터 타입
  • ✓ Fast access O(1) / 빠른 접근
  • ✓ Stack or heap allocation / 스택 또는 힙 할당

🟢 Python Lists / Python 리스트

  • ✓ Dynamic size / 동적 크기
  • ✓ References to objects / 객체 참조
  • ✓ Mixed data types / 혼합 데이터 타입
  • ✓ Fast access O(1) / 빠른 접근
  • ✓ Automatic memory management / 자동 메모리 관리

Declaration and Initialization

선언 및 초기화

// Method 1: Fixed size declaration / 고정 크기 선언
int arr[5];  // Uninitialized array / 초기화되지 않은 배열

// Method 2: Declaration with initialization / 초기화와 함께 선언
int numbers[5] = {10, 20, 30, 40, 50};

// Method 3: Auto size detection / 자동 크기 감지
int values[] = {1, 2, 3, 4, 5};

// Method 4: Partial initialization / 부분 초기화
int partial[5] = {1, 2};  // Rest are 0 / 나머지는 0
# Method 1: Empty list / 빈 리스트
arr = []

# Method 2: List with initial values / 초기값이 있는 리스트
numbers = [10, 20, 30, 40, 50]

# Method 3: List with mixed types / 혼합 타입 리스트
mixed = [1, "hello", 3.14, True]

# Method 4: List with repeated values / 반복 값으로 리스트 생성
zeros = [0] * 5  # [0, 0, 0, 0, 0]

Add Operation

추가 연산

#include <iostream>
using namespace std;

int main() {
    // C++ arrays have fixed size / C++ 배열은 고정 크기입니다
    // To "add", we need to create a new larger array / "추가"하려면 새 큰 배열이 필요
    
    int arr[5] = {10, 20, 30, 40, 50};
    int size = 5;
    
    // Create new array with size+1 / 크기+1인 새 배열 생성
    int newArr[6];
    
    // Copy old elements / 기존 요소 복사
    for(int i = 0; i < size; i++) {
        newArr[i] = arr[i];
    }
    
    // Add new element / 새 요소 추가
    newArr[5] = 60;
    
    // Alternative: Use std::vector for dynamic size / 대안: 동적 크기용 vector 사용
    // #include <vector>
    // vector<int> vec = {10, 20, 30, 40, 50};
    // vec.push_back(60);  // Much easier! / 훨씬 쉽습니다!
    
    return 0;
}

Note / 참고:

C++ arrays cannot grow dynamically.
Use std::vector for dynamic arrays.
C++ 배열은 동적으로 늘어날 수 없습니다.
동적 배열은 std::vector를 사용하세요.
# Method 1: append() - Add to end / 끝에 추가
numbers = [10, 20, 30, 40, 50]
numbers.append(60)
print(numbers)  # [10, 20, 30, 40, 50, 60]

# Method 2: insert() - Add at specific position / 특정 위치에 추가
numbers.insert(2, 25)  # Insert 25 at index 2 / 인덱스 2에 25 삽입
print(numbers)  # [10, 20, 25, 30, 40, 50, 60]

# Method 3: extend() - Add multiple items / 여러 항목 추가
numbers.extend([70, 80])

# Method 4: + operator / + 연산자
numbers = numbers + [90, 100]
print(numbers)

Output / 출력:

[10, 20, 25, 30, 40, 50, 60]

Update Operation

업데이트 연산

#include <iostream>
using namespace std;

int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    
    // Update single element / 단일 요소 업데이트
    arr[2] = 35;  // Change 30 to 35 / 30을 35로 변경
    
    cout << "Updated array: ";
    for(int i = 0; i < 5; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    // Update multiple elements / 여러 요소 업데이트
    for(int i = 0; i < 5; i++) {
        arr[i] = arr[i] * 2;  // Double each value / 각 값 2배로
    }
    
    cout << "After doubling: ";
    for(int i = 0; i < 5; i++) {
        cout << arr[i] << " ";
    }
    
    return 0;
}

Output / 출력:

Updated array: 10 20 35 40 50
After doubling: 20 40 70 80 100
# Update single element / 단일 요소 업데이트
numbers = [10, 20, 30, 40, 50]
numbers[2] = 35  # Change 30 to 35 / 30을 35로 변경
print("Updated list:", numbers)

# Update using negative index / 음수 인덱스로 업데이트
numbers[-1] = 55  # Last element / 마지막 요소
print("Updated last:", numbers)

# Update multiple elements using slice / 슬라이스로 여러 요소 업데이트
numbers[1:4] = [25, 35, 45]
print("Updated slice:", numbers)

# Update all elements / 모든 요소 업데이트
numbers = [x * 2 for x in numbers]  # List comprehension / 리스트 컴프리헨션
print("After doubling:", numbers)

Output / 출력:

Updated list: [10, 20, 35, 40, 50]
Updated last: [10, 20, 35, 40, 55]
Updated slice: [10, 25, 35, 45, 55]
After doubling: [20, 50, 70, 90, 110]

Search Operation

검색 연산

#include <iostream>
using namespace std;

// Linear search function / 선형 검색 함수
int linearSearch(int arr[], int size, int target) {
    for(int i = 0; i < size; i++) {
        if(arr[i] == target) {
            return i;  // Return index / 인덱스 반환
        }
    }
    return -1;  // Not found / 찾지 못함
}

int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    int size = 5;
    
    // Search for element / 요소 검색
    int target = 30;
    int index = linearSearch(arr, size, target);
    
    if(index != -1) {
        cout << "Found " << target << " at index " << index << endl;
    } else {
        cout << target << " not found" << endl;
    }
    
    // Check if element exists / 요소 존재 확인
    bool exists = (index != -1);
    cout << "Exists: " << (exists ? "Yes" : "No") << endl;
    
    return 0;
}

Output / 출력:

Found 30 at index 2
Exists: Yes

Count Operation

개수 세기 연산

#include <iostream>
using namespace std;

// Count occurrences / 발생 횟수 세기
int countOccurrences(int arr[], int size, int target) {
    int count = 0;
    for(int i = 0; i < size; i++) {
        if(arr[i] == target) {
            count++;
        }
    }
    return count;
}

int main() {
    int arr[] = {10, 20, 30, 20, 40, 20, 50};
    int size = 7;
    
    // Count specific value / 특정 값 개수
    int count20 = countOccurrences(arr, size, 20);
    cout << "20 appears " << count20 << " times" << endl;
    
    // Count elements greater than value / 특정 값보다 큰 요소
    int countGreater = 0;
    for(int i = 0; i < size; i++) {
        if(arr[i] > 25) countGreater++;
    }
    cout << "Elements > 25: " << countGreater << endl;
    
    // Total array size / 배열 크기
    cout << "Array size: " << size << endl;
    
    return 0;
}

Output / 출력:

20 appears 3 times
Elements > 25: 3
Array size: 7
numbers = [10, 20, 30, 20, 40, 20, 50]

# Count specific value / 특정 값 개수
count20 = numbers.count(20)
print(f"20 appears {count20} times")

# Count elements with condition / 조건에 맞는 요소 개수
count_greater = sum(1 for x in numbers if x > 25)
print(f"Elements > 25: {count_greater}")

# Total list size / 리스트 크기
print(f"List size: {len(numbers)}")

# Count unique elements / 고유 요소 개수
unique_count = len(set(numbers))
print(f"Unique elements: {unique_count}")

# Count all occurrences using Counter / Counter로 모든 발생 횟수
from collections import Counter
counts = Counter(numbers)
print(f"All counts: {dict(counts)}")

Output / 출력:

20 appears 3 times
Elements > 25: 3
List size: 7
Unique elements: 5
All counts: {10: 1, 20: 3, 30: 1, 40: 1, 50: 1}

🎯 Key Takeaways / 핵심 요점

  • C++ arrays: Fast, fixed size, memory efficient / 빠름, 고정 크기, 메모리 효율적
  • Python lists: Flexible, dynamic, easier to use / 유연함, 동적, 사용하기 쉬움
  • Use C++ arrays when: Size is known, performance critical / 크기를 알 때, 성능이 중요할 때
  • Use Python lists when: Need flexibility, rapid development / 유연성이 필요할 때, 빠른 개발
